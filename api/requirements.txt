# Pinned exactly, not `>=`. Every one of these was a floor that had drifted years
# away from what actually runs: `TikTokLive>=0.8.0` sat under a 7.0.1 install, so
# the floor described a version nobody had used since long before this repo
# existed. A floor that far below reality is not a constraint, it is a comment —
# and `pip install -r requirements.txt` on a fresh machine would happily resolve
# to something untested. These are the versions the suites pass against.
fastapi==0.143.0
uvicorn[standard]==0.54.0
TikTokLive==7.0.1
websockets==17.2
pydantic==2.14.0
# `logging_setup.py` imports this to format every log line as JSON, and the import
# is wrapped in `try/except ImportError` so a missing package is not a crash — it
# silently falls back to plain text. That is exactly why it has to be declared:
# the fallback is meant for a machine that has not been set up, not for this one,
# and the difference is invisible in a log file until somebody greps for a field
# that is not there. It was installed but never listed.
python-json-logger==4.2.0
# YouTube Live source. Installed from the fork's git URL, not PyPI: upstream
# 0.5.5 generates chat continuations with a protobuf YouTube stopped accepting
# and returns *zero items with no error*. The fork (v0.6.0, tag on
# aldifhr/pytchatV2) scrapes the page's own token instead. A bare
# `pytchat==0.6.0` would not resolve — PyPI has no such version — so the URL is
# the only form that installs on a fresh machine.
pytchat @ git+https://github.com/aldifhr/pytchatV2.git@v0.6.0
# pytchat declares `httpx[http2]` itself, but fastapi/starlette already pull
# httpx for TestClient; pinning the range here keeps the two from disagreeing.
httpx[http2]>=0.27,<0.29
